cmake_minimum_required(VERSION 3.16)
project(conavbot_sim)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(ignition-gazebo6 REQUIRED)
find_package(ignition-plugin1 REQUIRED COMPONENTS register)
add_library(conavbot_motor SHARED cpp/motor_system.cc)
target_compile_features(conavbot_motor PUBLIC cxx_std_17)
target_link_libraries(conavbot_motor ignition-gazebo6::ignition-gazebo6 ignition-plugin1::register)
install(TARGETS conavbot_motor LIBRARY DESTINATION lib)
ament_python_install_package(${PROJECT_NAME})
install(DIRECTORY config launch assets DESTINATION share/${PROJECT_NAME})
ament_package()
